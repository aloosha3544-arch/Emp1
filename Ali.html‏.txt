    <!-- جافاسكريبت للتحكم والوظائف -->    <script>
        const allowedEmployeeIds = ["750158", "741019", "812714", "802808", "810061", "827118", "832421", "796026", "808083", "300958", "301628"];
        const allowedManagerIds = ["750158", "741019", "832421", "796026"];

        let currentRole = '';
        let isRegisterMode = false;
        let loggedInUserId = '';
        let usersDB = JSON.parse(localStorage.getItem('system_users')) || {};
        let employeeRecords = JSON.parse(localStorage.getItem('employee_records')) || [];
        let managerDirectives = JSON.parse(localStorage.getItem('manager_directives')) || [];

        function openLoginModal(role) {
            currentRole = role;
            isRegisterMode = false;
            updateModalUI();
            document.getElementById('auth-modal').classList.remove('hidden');
        }

        function closeModal() {
            document.getElementById('auth-modal').classList.add('hidden');
            document.getElementById('auth-form').reset();
        }

        function toggleAuthMode() {
            isRegisterMode = !isRegisterMode;
            updateModalUI();
        }

        function updateModalUI() {
            const title = document.getElementById('modal-title');
            const submitBtn = document.getElementById('auth-submit-btn');
            const toggleBtn = document.getElementById('toggle-mode-btn');
            const roleName = currentRole === 'employee' ? 'الموظفين' : 'المدير';

            if (isRegisterMode) {
                title.innerText = `تخصيص رمز سري جديد (${roleName})`;
                submitBtn.innerText = 'حفظ الرمز والبدء';
                toggleBtn.innerText = 'لديك رمز سري بالفعل؟ تسجيل الدخول';
            } else {
                title.innerText = `تسجيل الدخول (${roleName})`;
                submitBtn.innerText = 'دخول';
                toggleBtn.innerText = 'ليس لديك حساب؟ تخصيص رمز سري جديد';
            }
        }

        function handleAuth(event) {
            event.preventDefault();
            const id = document.getElementById('auth-id').value.trim();
            const pass = document.getElementById('auth-pass').value.trim();

            if (currentRole === 'employee' && !allowedEmployeeIds.includes(id)) {
                alert('عذراً، هذا الرقم الوظيفي غير مصرح له بالدخول إلى خانة الموظفين!');
                return;
            }

            if (currentRole === 'manager' && !allowedManagerIds.includes(id)) {
                alert('عذراً، هذا الرقم الوظيفي غير مخول بالدخول إلى خانة المدير!');
                return;
            }

            if (isRegisterMode) {
                if (usersDB[id]) {
                    alert('هذا الرقم الوظيفي لديه رمز سري مسجل مسبقاً! قم بتسجيل الدخول مباشرة.');
                    return;
                }
                usersDB[id] = { pass: pass, role: currentRole };
                localStorage.setItem('system_users', JSON.stringify(usersDB));
                alert('تم تخصيص الرمز السري الخاص بك بنجاح! يمكنك الدخول به الآن.');
                toggleAuthMode();
                return;
            } else {
                if (!usersDB[id]) {
                    alert('لم يتم العثور على رمز سري مسجل لهذا الرقم الوظيفي. يرجى اختيار "تخصيص رمز سري جديد" أولاً.');
                    return;
                }

                if (usersDB[id].pass !== pass || usersDB[id].role !== currentRole) {
                    alert('الرقم السري غير صحيح، أو أن هذا الرقم الوظيفي مسجل في خانة أخرى!');
                    return;
                }
            }

            loggedInUserId = id;
            closeModal();
            
            document.getElementById('main-container').classList.add('hidden');

            if (currentRole === 'employee') {
                document.getElementById('employee-dashboard').classList.remove('hidden');
                document.getElementById('emp-id-field').value = id;
                document.getElementById('emp-date').valueAsDate = new Date();
                
                const lastEmpRecord = employeeRecords.slice().reverse().find(r => r.id === id);
                if (lastEmpRecord) {
                    document.getElementById('emp-name').value = lastEmpRecord.name;
                    document.getElementById('emp-section').value = lastEmpRecord.section;
                }
                loadEmployeeMyRecords();
                loadEmployeeDirectives(); 
            } else {
                document.getElementById('manager-dashboard').classList.remove('hidden');
                loadManagerTable(employeeRecords);
                loadManagerDirectives();
            }
        }

        function logout() {

        function handleManagerDirective(e) {
            e.preventDefault();
            const title = document.getElementById('manager-directive-title').value;
            const fileInput = document.getElementById('manager-directive-file');
            
            if (fileInput.files.length === 0) {
                alert('يرجى اختيار ملف (صورة أو إكسل)');
                return;
            }

            const file = fileInput.files[0];
            const reader = new FileReader();

            reader.onload = function(uploadEvent) {
                const directive = {
                    date: new Date().toISOString().split('T')[0],
                    title: title,
                    fileName: file.name,
                    fileType: file.type,
                    fileData: uploadEvent.target.result
                };

                managerDirectives.push(directive);
                localStorage.setItem('manager_directives', JSON.stringify(managerDirectives));
                alert('تم نشر التعميم والمرفق بنجاح!');
                document.getElementById('manager-directive-title').value = '';
                fileInput.value = '';
                loadManagerDirectives();
            };

            reader.readAsDataURL(file);
        }

        function loadManagerDirectives() {
            const tbody = document.getElementById('manager-directives-table');
            if (!tbody) return;

        }

        // دالة موحدة لتوليد زر المعاينة للصور أو التحميل للإكسل
        function renderFileLink(item) {
            if (!item || !item.fileData) return `<span class="text-gray-400 text-xs">بدون مرفق</span>`;
            
            const isImage = item.fileType && item.fileType.startsWith('image/');
            const fileName = item.fileName || 'ملف مرفق';

            if (isImage) {
                return `<button onclick="openImagePreview('${item.fileData}', '${item.title || fileName}')" class="bg-blue-50 text-blue-600 hover:bg-blue-100 px-3 py-1.5 rounded-xl text-xs font-semibold transition inline-flex items-center gap-1.5"><i class="fa-solid fa-image"></i> معاينة الصورة (${fileName})</button>`;
            } else {
                return `<a href="${item.fileData}" download="${fileName}" class="bg-emerald-50 text-emerald-600 hover:bg-emerald-100 px-3 py-1.5 rounded-xl text-xs font-semibold transition inline-flex items-center gap-1.5"><i class="fa-solid fa-file-excel"></i> تحميل ملف الإكسل (${fileName})</a>`;
            }
        }

        // التعامل مع إرسال سجل الموظف مع دعم رفع الملفات (صورة أو إكسل)
        document.getElementById('employee-record-form').addEventListener('submit', function (e) {
            e.preventDefault();
            const editIndex = parseInt(document.getElementById('edit-index').value);
            const name = document.getElementById('emp-name').value.trim();
            const section = document.getElementById('emp-section').value;
            const attendance = document.getElementById('emp-attendance').value;
            const task = document.getElementById('emp-task').value.trim();
            const date = document.getElementById('emp-date').value;
            const fileInput = document.getElementById('emp-attachment');

            // دالة داخلية لحفظ البيانات بعد قراءة الملف إن وجد
            function saveRecordWithFile(fileData, fileName, fileType) {
                let finalFileName = fileName;
                let finalFileData = fileData;
                let finalFileType = fileType;

                if (!fileData && editIndex !== -1) {
                    finalFileName = employeeRecords[editIndex].fileName;
                    finalFileData = employeeRecords[editIndex].fileData;
                    finalFileType = employeeRecords[editIndex].fileType;
                }

                const recordData = {
                    id: loggedInUserId,
                    name: name,
                    section: section,
                    attendance: attendance,
                    task: task,
                    date: date,
                    fileName: finalFileName,
                    fileData: finalFileData,
                    fileType: finalFileType,
                    attachmentName: finalFileName // للحفاظ التوافقية مع جداول التصدير
                };

                if (editIndex === -1) {
                    employeeRecords.push(recordData);
                    alert('تم حفظ وإرسال البيانات اليومية بنجاح!');
                } else {
                    employeeRecords[editIndex] = recordData;
                    alert('تم تحديث السجل بنجاح!');
                    cancelEdit();
                }

                localStorage.setItem('employee_records', JSON.stringify(employeeRecords));
                loadEmployeeMyRecords();
                document.getElementById('employee-record-form').reset();
                document.getElementById('emp-id-field').value = loggedInUserId;
                document.getElementById('emp-date').valueAsDate = new Date();
            }

            if (fileInput.files.length > 0) {
                const file = fileInput.files[0];
                const reader = new FileReader();
                reader.onload = function(uploadEvent) {
                    saveRecordWithFile(uploadEvent.target.result, file.name, file.type);
                };
                reader.readAsDataURL(file);
            } else {
                saveRecordWithFile(null, null, null);
            }
        });

        function exportToCSV(data, filename) {
            let csvContent = "\uFEFF"; 
            csvContent += "التاريخ,الرقم الوظيفي,اسم الموظف,الشعبة / الوحدة,الحالة,العمل المكلف به,المرفق\n";

            data.forEach(row => {
                const cleanTask = `"${(row.task || '').replace(/"/g, '""')}"`;
                const cleanName = `"${(row.name || '').replace(/"/g, '""')}"`;
                const cleanSection = `"${(row.section || '').replace(/"/g, '""')}"`;
                const line = [row.date, row.id, cleanName, cleanSection, row.attendance, cleanTask, row.fileName || 'لا يوجد'].join(",");
                csvContent += line + "\n";
            });

            const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.setAttribute('href', url);
            a.setAttribute('download', filename);
            document.body.appendChild(a);
            a.click();
            document.body.removeChild(a);
        }
